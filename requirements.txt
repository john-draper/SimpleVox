# SimpleVox — Python dependencies
#
# IMPORTANT: Install PyTorch FIRST and SEPARATELY for your CUDA version.
# For CUDA 11.8 (RTX 3050/3060/3070/3080/3090, A-series):
#   pip install torch==2.5.1 torchaudio==2.5.1 --index-url https://download.pytorch.org/whl/cu118
# For other GPUs/CPU-only, see: https://pytorch.org/get-started/locally/
#
# Then install the remaining dependencies:
#   pip install -r requirements.txt
#
# ffmpeg must also be installed system-wide (see README.md).

# ASR: one model, one pass. faster-whisper emits word-level timestamps itself,
# so there is no alignment stage and no WhisperX requirement.
# (WhisperX is NOT installed by this file; the legacy batch_censor.py still
# imports it, but run.py / batch_videos.py / transcribe.py no longer do.)
faster-whisper>=1.2.0
torchaudio>=2.1.0

# edge-tts: free Microsoft Edge neural TTS (no API key, reliable).
# This is the ONLY TTS provider SimpleVox uses — one generic male voice for all
# replacement words.
edge-tts>=6.1.0

# Second ear (--second-ear / ensemble_audit.py): NVIDIA Parakeet.
# Parakeet-CTC works with transformers >= 4.57.
# Parakeet-TDT (preferred: native word timestamps + punctuation) needs
# transformers >= 5.9 — SimpleVox falls back to CTC automatically when TDT is
# not importable.
transformers>=4.57
librosa
soundfile
